var pagamento
var bandeira


var pagamento = prompt("Digite a forma de pagamento:");

if (pagamento == "Dinheiro") {
    alert("Pagamento em dinheiro não possui parcelamento.");
} else {
    var bandeira = prompt("Digite a bandeira do cartão:");

    if (bandeira == "Cartão da Loja") {
        alert("Parcelamento disponível em até 12x sem juros.");
    } else {
        alert("Parcelamento disponível em até 6x.");
    }
}
